cmake_minimum_required(VERSION 3.20)
project(DrinkKDE LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror)

find_package(Qt6 6.7 REQUIRED COMPONENTS Widgets DBus)
find_package(LayerShellQt 6.6 REQUIRED)
find_package(KF6Service REQUIRED)
find_package(KF6StatusNotifierItem 6.14 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PipeWire REQUIRED IMPORTED_TARGET libpipewire-0.3)

# Both native C++ adapters consume the same generated interface. Let Cargo
# handle incremental rebuilds of the scheduler, bridge and locked dependencies.
set(cargo_target "${CMAKE_BINARY_DIR}/cargo")
set(bridge "${cargo_target}/release/libdrink_macos_bridge.a")
add_custom_target(scheduler
    COMMAND cargo build --locked --release -p drink-macos-bridge --target-dir "${cargo_target}"
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}/.."
    BYPRODUCTS "${bridge}"
    VERBATIM)

add_library(drink-audio STATIC CaptureGraph.cpp Microphone.cpp)
target_link_libraries(drink-audio PRIVATE PkgConfig::PipeWire)
target_include_directories(drink-audio PUBLIC "${PROJECT_SOURCE_DIR}")

add_library(drink-plasma STATIC kde/AppImageUpdate.cpp kde/BreakWindows.cpp kde/Drink.cpp kde/Foreground.cpp
    kde/Instance.cpp kde/Session.cpp)
set_target_properties(drink-plasma PROPERTIES AUTOMOC ON)
add_dependencies(drink-plasma scheduler)
target_include_directories(drink-plasma PUBLIC kde "${cargo_target}/cxxbridge")
target_link_libraries(drink-plasma PUBLIC drink-audio Qt6::Widgets Qt6::DBus KF6::StatusNotifierItem
    PRIVATE LayerShellQt::Interface KF6::Service "${bridge}" ${CMAKE_DL_LIBS})

function(drink_executable target version)
    add_executable(${target} kde/main.cpp kde/resources.qrc)
    set_target_properties(${target} PROPERTIES AUTORCC ON INSTALL_RPATH_USE_LINK_PATH ON)
    target_link_libraries(${target} PRIVATE drink-plasma)
    if(NOT "${version}" STREQUAL "")
        target_compile_definitions(${target} PRIVATE DRINK_RELEASE_VERSION="${version}")
    endif()
endfunction()

set(DRINK_RELEASE_VERSION "" CACHE STRING "Release date (YYYY.MM.DD); an empty value builds Drink Dev")
option(DRINK_APPIMAGE "Enable signed self-updates for the packaged x86-64 AppImage" OFF)
if(DRINK_APPIMAGE AND (DRINK_RELEASE_VERSION STREQUAL "" OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64"))
    message(FATAL_ERROR "DRINK_APPIMAGE requires an x86-64 dated release")
endif()
if(DRINK_RELEASE_VERSION STREQUAL "")
    set(drink_program drink-kde-dev)
    set(drink_name "Drink Dev")
    set(drink_id io.github.napolitain.Drink.Dev)
else()
    if(NOT DRINK_RELEASE_VERSION MATCHES "^[0-9][0-9][0-9][0-9]\\.[0-9][0-9]\\.[0-9][0-9]$")
        message(FATAL_ERROR "DRINK_RELEASE_VERSION must be YYYY.MM.DD")
    endif()
    set(drink_program drink-kde)
    set(drink_name Drink)
    set(drink_id io.github.napolitain.Drink)
endif()
drink_executable(${drink_program} "${DRINK_RELEASE_VERSION}")
if(DRINK_APPIMAGE)
    target_compile_definitions(${drink_program} PRIVATE DRINK_APPIMAGE)
    target_sources(${drink_program} PRIVATE kde/appimage.qrc)
endif()
configure_file(kde/drink.desktop.in "${drink_id}.desktop" @ONLY)
include(GNUInstallDirs)
install(TARGETS ${drink_program} RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${CMAKE_BINARY_DIR}/${drink_id}.desktop" DESTINATION "${CMAKE_INSTALL_DATADIR}/applications")
install(FILES kde/drink.svg DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps"
    RENAME "${drink_id}.svg")

include(CTest)
if(BUILD_TESTING)
    if(DRINK_RELEASE_VERSION STREQUAL "")
        drink_executable(drink-kde "2000.01.01")
        set(counterpart drink-kde)
    else()
        drink_executable(drink-kde-dev "")
        set(counterpart drink-kde-dev)
    endif()
    set_target_properties(${counterpart} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-programs")
    add_executable(capture-test tests/capture.cpp)
    target_link_libraries(capture-test PRIVATE drink-audio)
    add_test(NAME capture-graph COMMAND capture-test)
    if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
        add_executable(appimage-update-test tests/appimage-update.cpp)
        target_link_libraries(appimage-update-test PRIVATE drink-plasma)
        add_test(NAME appimage-update COMMAND bash "${PROJECT_SOURCE_DIR}/tests/appimage-update"
            "$<TARGET_FILE:appimage-update-test>" "$<TARGET_FILE:Qt6::rcc>")
    endif()
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(plasma-test tests/plasma.cpp kde/resources.qrc)
    set_target_properties(plasma-test PROPERTIES AUTOMOC ON AUTORCC ON)
    target_link_libraries(plasma-test PRIVATE drink-plasma Qt6::Test)
    target_compile_definitions(plasma-test PRIVATE DRINK_TEST_PROGRAM="$<TARGET_FILE:${drink_program}>")
    add_test(NAME plasma-behavior COMMAND plasma-test)
    add_test(NAME plasma-handoff COMMAND "${PROJECT_SOURCE_DIR}/tests/lifecycle"
        "$<TARGET_FILE:drink-kde-dev>" "$<TARGET_FILE:drink-kde>")
    set_tests_properties(plasma-behavior plasma-handoff PROPERTIES
        LABELS plasma RESOURCE_LOCK plasma-session TIMEOUT 90)
endif()
